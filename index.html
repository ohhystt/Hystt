        const container = document.querySelector('.roadmap-container');

        const svgPath = document.getElementById('path-line');

        const plane = document.getElementById('plane');

        const cards = document.querySelectorAll('.step-card');

        

        let pathPoints = [];


        function drawPath() {

            if (cards.length === 0) return;


            const containerRect = container.getBoundingClientRect();

            let pathD = "";

            pathPoints = []; 


            cards.forEach((card) => {

                const rect = card.getBoundingClientRect();

                const centerX = rect.left + rect.width / 2 - containerRect.left;

                const centerY = rect.top + rect.height / 2 - containerRect.top;

                pathPoints.push({ x: centerX, y: centerY });

            });


            pathD += `M ${pathPoints[0].x} ${pathPoints[0].y}`;


            for (let i = 0; i < pathPoints.length - 1; i++) {

                const p1 = pathPoints[i];

                const p2 = pathPoints[i+1];

                const distY = p2.y - p1.y;

                const controlY = distY * 0.5; 

                pathD += ` C ${p1.x} ${p1.y + controlY}, ${p2.x} ${p2.y - controlY}, ${p2.x} ${p2.y}`;

            }


            svgPath.setAttribute('d', pathD);

        }


        function animatePlane() {

            if (pathPoints.length === 0) return;


            const path = document.getElementById('path-line');

            const pathLength = path.getTotalLength();

            const containerRect = container.getBoundingClientRect();

            const windowHeight = window.innerHeight;

            

            const firstPointY = pathPoints[0].y;

            const lastPointY = pathPoints[pathPoints.length - 1].y;

            

            const containerTopRelativeToCenter = windowHeight / 2 - containerRect.top;

            const travelDistance = lastPointY - firstPointY;

            const currentProgress = containerTopRelativeToCenter - firstPointY;


            let percentage = currentProgress / travelDistance;


            // Force le 100% à la fin

            const scrollPosition = window.scrollY + window.innerHeight;

            const bodyHeight = document.body.scrollHeight;

            

            if (scrollPosition >= bodyHeight - 50) {

                 percentage = 1;

            }


            percentage = Math.max(0, Math.min(1, percentage));


            const point = path.getPointAtLength(percentage * pathLength);

            

            const lookAheadDist = 10;

            let p1, p2;

            if (percentage < 1.0) {

                p1 = point;

                p2 = path.getPointAtLength(Math.min((percentage * pathLength) + lookAheadDist, pathLength));

            } else {

                p1 = path.getPointAtLength(pathLength - lookAheadDist);

                p2 = point;

            }

            const angle = Math.atan2(p2.y - p1.y, p2.x - p1.x) * 180 / Math.PI;


            plane.style.left = `${point.x}px`;

            plane.style.top = `${point.y}px`;

            plane.style.transform = `translate(-50%, -50%) rotate(${angle + 90}deg)`; 

            

            cards.forEach((card, index) => {

                if (index === cards.length - 1 && percentage >= 0.95) {

                    card.classList.add('active');

                    return;

                }


                const rect = card.getBoundingClientRect();

                const cardCenterY = rect.top + rect.height/2;

                const planeAbsoluteY = containerRect.top + point.y;

                const dist = Math.abs(cardCenterY - planeAbsoluteY);

                

                if (dist < 250) {

                    card.classList.add('active');

                } else {

                    card.classList.remove('active');

                }

            });

        }


        // CORRECTION CRITIQUE : ResizeObserver pour recalculer le chemin si la taille du contenu change (chargement images)

        const resizeObserver = new ResizeObserver(() => {

            drawPath();

            animatePlane();

        });

        resizeObserver.observe(container);


        window.addEventListener('resize', () => { drawPath(); animatePlane(); });

        window.addEventListener('scroll', () => { requestAnimationFrame(animatePlane); });

        window.addEventListener('DOMContentLoaded', () => {

            // Timeout pour être sûr

            setTimeout(() => { drawPath(); animatePlane(); }, 100);

        });


    </script>
